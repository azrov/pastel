#Pastel.Web

HTTP server for web APIs and sites.

Web := TWebServer.Create(8080);
Web.Get('/hello', @Hello);
Web.Get('/users/:id', @GetUser);
Web.Post('/users', @CreateUser);
Web.Static('/', 'public');           // files: public/index.html, public/app.css, ...
Web.Health('/health', AppVersion);
Web.Start;                           // serves in the background until Web.Stop

procedure GetUser(Req: TWebRequest; Res: TWebResponse);
var
  User: TJSONObject;
begin
  User := FindUser(Req.ParamInt('id'));
  if User = nil then
    raise EWebError.Create(404, 'no user with this id');
  Res.Json(User);
end;

Requests are handled at the same time in several threads: what handlers share must be protected (TCriticalSection) or be read-only. Errors become JSON answers: EWebError(404, 'no user') -> 404 {"error": "no user"}; a bad JSON body -> 400; any other exception -> 500, logged, with the message in the answer only in debug builds.
