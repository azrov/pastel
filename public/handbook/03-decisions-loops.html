#3. Decisions and loops

A program that always does the same thing is not much use. This chapter makes programs choose (if, case) and repeat (for, while, repeat).

#if

program Weather;

{$mode objfpc}{$H+}

var
  Celsius: Integer;
begin
  Celsius := 31;
  if Celsius > 30 then
    WriteLn('Hot. Drink water.');
  if Celsius < 0 then
    WriteLn('Freezing.')
  else
    WriteLn('Not freezing.');
end.

Hot. Drink water.
Not freezing.

No semicolon before else. if … then A else B is one statement; a semicolon after A would end it, and the else would stand alone — the compiler stops there.

#Several statements: begin … end

then and else take one statement. For more, put them between begin and end — a compound statement:

program Withdraw;

{$mode objfpc}{$H+}

var
  Balance, Amount: Integer;
begin
  Balance := 100;
  Amount := 130;
  if Amount <= Balance then
  begin
    Balance := Balance - Amount;
    WriteLn('Here you are.');
  end
  else
  begin
    WriteLn('Not enough money.');
    WriteLn('You have ', Balance, '.');
  end;
end.

Not enough money.
You have 100.

#else if

Choices in a row are written as a chain:

program Grades;

{$mode objfpc}{$H+}

var
  Score: Integer;
begin
  Score := 74;
  if Score >= 85 then
    WriteLn('Very good')
  else if Score >= 70 then
    WriteLn('Good')
  else if Score >= 50 then
    WriteLn('Passed')
  else
    WriteLn('Failed');
end.

Good

The first condition that holds wins; the others are not even looked at.

#case

When one value decides between many choices, case is clearer than a chain of ifs. It works for whole numbers, characters, enumerations (chapter 5) and — in Free Pascal — strings:

program Days;

{$mode objfpc}{$H+}

var
  Day: Integer;
  Command: string;
begin
  Day := 6;
  case Day of
    1..5: WriteLn('A working day');
    6, 7: WriteLn('Weekend!');
  else
    WriteLn('There is no day ', Day);
  end;

  Command := 'list';
  case Command of
    'add': WriteLn('Adding...');
    'list', 'ls': WriteLn('Listing...');
  else
    WriteLn('Unknown command: ', Command);
  end;
end.

Weekend!
Listing...

1..5 is a range, 6, 7 a list. The else part runs when nothing matched; without it, nothing happens. Every case ends with its own end.

#for

for repeats a statement for every number from one value to another:

program Counting;

{$mode objfpc}{$H+}

var
  I, Sum: Integer;
begin
  for I := 1 to 5 do
    Write(I, ' ');
  WriteLn;

  for I := 3 downto 1 do
    WriteLn(I, '...');
  WriteLn('Go!');

  Sum := 0;
  for I := 1 to 100 do
    Sum := Sum + I;
  WriteLn('1 + 2 + ... + 100 = ', Sum);
end.

1 2 3 4 5
3...
2...
1...
Go!
1 + 2 + ... + 100 = 5050

to counts up, downto down. The loop variable must be a local variable of a whole-number type (or a Char, or an enumeration); do not change it inside the loop, and do not rely on its value after the loop ends — the compiler does not promise what it is.

Loops can be inside each other:

program Table;

{$mode objfpc}{$H+}

uses
  SysUtils;

var
  Row, Col: Integer;
begin
  for Row := 1 to 4 do
  begin
    for Col := 1 to 4 do
      Write(Format('%4d', [Row * Col]));
    WriteLn;
  end;
end.

   1   2   3   4
   2   4   6   8
   3   6   9  12
   4   8  12  16

#while and repeat

for is for a known number of times. When the loop goes on until something happens, use while or repeat:

program Savings;

{$mode objfpc}{$H+}

uses
  SysUtils;

var
  Balance: Double;
  Years: Integer;
begin
  { how many years until 1000 grows to 2000 at 7% a year? }
  Balance := 1000;
  Years := 0;
  while Balance < 2000 do
  begin
    Balance := Balance * 1.07;
    Inc(Years);
  end;
  WriteLn(Format('%d years: %.2f', [Years, Balance]));
end.

11 years: 2104.85

while checks before each round, so its body may never run. repeat … until checks after, so the body runs at least once — the natural form for "ask until the answer is right":

program AskNumber;

{$mode objfpc}{$H+}

uses
  SysUtils;

var
  S: string;
  N: Integer;
begin
  repeat
    Write('A number from 1 to 10: ');
    ReadLn(S);
  until TryStrToInt(S, N) and (N >= 1) and (N <= 10);
  WriteLn('Thank you: ', N);
end.

twelve
12
7

A number from 1 to 10: A number from 1 to 10: A number from 1 to 10: Thank you: 7

repeat and until hold several statements without begin/end. Note that until takes the condition to stop, while the condition to go on.

#Break and Continue

Break leaves the loop at once; Continue skips to the next round:

program Search;

{$mode objfpc}{$H+}

var
  I: Integer;
begin
  { the first number above 100 that 7 and 11 both divide }
  for I := 101 to 1000 do
    if (I mod 7 = 0) and (I mod 11 = 0) then
    begin
      WriteLn('Found: ', I);
      Break;
    end;

  { the odd numbers only }
  for I := 1 to 9 do
  begin
    if I mod 2 = 0 then
      Continue;
    Write(I, ' ');
  end;
  WriteLn;
end.

Found: 154
1 3 5 7 9

#In Pocket

A first look at what Pocket will do: the expenses of a week, the days over a daily budget, the total. (The amounts are written straight into the program here; chapter 5 keeps them in an array.)

program Week;

{$mode objfpc}{$H+}

uses
  SysUtils;

const
  DailyBudget = 40;

var
  Day: Integer;
  Spent, Total: Currency;
begin
  Total := 0;
  for Day := 1 to 7 do
  begin
    case Day of
      1: Spent := 32.50;
      2: Spent := 18;
      3: Spent := 55.20;
      4: Spent := 12;
      5: Spent := 41;
      6: Spent := 0;
    else
      Spent := 26.75;
    end;
    Total := Total + Spent;
    if Spent > DailyBudget then
      WriteLn(Format('Day %d: %.2f - over the budget', [Day, Double(Spent)]))
    else if Spent = 0 then
      WriteLn(Format('Day %d: nothing spent', [Day]));
  end;
  WriteLn(Format('The week: %.2f (a budget of %d)', [Double(Total), DailyBudget * 7]));
end.

Day 3: 55.20 - over the budget
Day 5: 41.00 - over the budget
Day 6: nothing spent
The week: 185.45 (a budget of 280)

#Exercises

  1. Print the numbers from 1 to 30, but Fizz for those divisible by 3, Buzz for 5, and FizzBuzz for both.
  2. Ask for a month number and print how many days it has (ignore leap years) with one case.
  3. Print a triangle of stars, five lines, one star more on each line.
  4. ★ The computer "thinks" of a number from 1 to 100 (Randomize; Secret := Random(100) + 1;). Let the user guess, and say "higher" or "lower" until it is right; then say how many guesses it took.
