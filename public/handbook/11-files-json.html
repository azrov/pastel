#11. Files and JSON

A program forgets everything when it ends — unless it writes it down. This chapter writes and reads files, and uses JSON, the text format nearly every program and web service understands. At the end, Pocket keeps its entries between runs.

#Text files the easy way

Pastel.Files reads and writes whole files in one call — always as UTF-8, on every system:

program Notes;

{$mode objfpc}{$H+}

uses
  SysUtils, Pastel.Files;

var
  Dir, FileName, Line: string;
begin
  Dir := MakeTempDir('notes');                     // a new empty folder for this example
  FileName := JoinPath([Dir, 'notes.txt']);
  WriteText(FileName, 'first line' + LineEnding);
  AppendText(FileName, 'second line' + LineEnding);
  WriteLn('Exists: ', FileExists(FileName), ', ', FileSize(FileName), ' bytes');
  for Line in ReadLines(FileName) do
    WriteLn('> ', Line);
  WriteLines(JoinPath([Dir, 'list.txt']), ['tea', 'coffee', 'cocoa']);
  WriteLn(Trim(ReadText(JoinPath([Dir, 'list.txt']))).Replace(LineEnding, ', '));
  DeleteDir(Dir);
  WriteLn('Cleaned up: ', not DirectoryExists(Dir));
end.

Exists: TRUE, 23 bytes
> first line
> second line
tea, coffee, cocoa
Cleaned up: TRUE
ReadText, WriteText, AppendText a whole file as one string
ReadLines, WriteLines as an array of lines
ReadBytes, WriteBytes any file, as bytes
JoinPath(['a', 'b', 'c.txt']) a path with the system's separator (/ or \)
EnsureDir, DeleteDir, ListFiles(Dir, '*.txt') folders
HomeDir, AppDir, MakeTempDir where things are
FileExists, DirectoryExists (SysUtils) is it there?

Build paths with JoinPath, never by adding '/': the same code then works on Windows.

#The classic way

Pascal's own text files read a line at a time — useful for files too big to read at once:

program Classic;

{$mode objfpc}{$H+}

uses
  SysUtils;

var
  F: TextFile;
  Line: string;
  Count: Integer;
begin
  AssignFile(F, 'lines.txt');
  Rewrite(F);                      // a new file (Append(F): add to one; Reset(F): read)
  WriteLn(F, 'alpha');
  WriteLn(F, 'beta');
  CloseFile(F);

  AssignFile(F, 'lines.txt');
  Reset(F);
  Count := 0;
  try
    while not Eof(F) do
    begin
      ReadLn(F, Line);
      Inc(Count);
      WriteLn(Count, ': ', Line);
    end;
  finally
    CloseFile(F);
  end;
  DeleteFile('lines.txt');
end.

1: alpha
2: beta

WriteLn(F, …) and ReadLn(F, …) are the screen's WriteLn and ReadLn, aimed at a file.

#JSON

JSON writes data as text that both people and programs can read:

{
  "name": "Pocket",
  "budget": 1500,
  "categories": ["Food", "Rent"],
  "owner": { "name": "Ada", "email": "ada@example.com" }
}

Objects in {} hold named values, arrays in [] hold lists; values are strings, numbers, true/false, null, or more objects and arrays.

Free Pascal's fpjson unit keeps JSON as objects: TJSONObject, TJSONArray, and values — all children of TJSONData. Pastel.Json adds short helpers that read values by a path:

program ReadJson;

{$mode objfpc}{$H+}

uses
  SysUtils, fpjson, Pastel.Json;

const
  Sample = '{"name": "Pocket", "budget": 1500, "categories": ["Food", "Rent"],' +
    ' "owner": {"name": "Ada", "email": "ada@example.com"}}';

var
  Data: TJSONData;
  C: TJSONData;
begin
  Data := ParseJson(Sample);
  try
    WriteLn('Name:   ', JStr(Data, 'name'));
    WriteLn('Budget: ', JInt(Data, 'budget'));
    WriteLn('Owner:  ', JStr(Data, 'owner.name'), ' <', JStr(Data, 'owner.email'), '>');
    WriteLn('First:  ', JStr(Data, 'categories[0]'));
    WriteLn('Phone:  "', JStr(Data, 'owner.phone', 'none'), '"');
    for C in JArray(Data, 'categories') do
      WriteLn('  - ', C.AsString);
  finally
    Data.Free;
  end;
end.

Name:   Pocket
Budget: 1500
Owner:  Ada <ada@example.com>
First:  Food
Phone:  "none"
  - Food
  - Rent
  • ParseJson turns text into objects — and you free what it gives back.
  • JStr, JInt, JFloat, JBool read a value by its path: names with dots, array elements with [0]. A missing value gives the default (the third parameter) instead of an error, which suits data from outside.
  • JArray gives the elements of an array, for a for … in loop.

#Writing JSON

JObject makes an object from names and values in one line; a TJSONArray collects objects. ToJson writes them as text, SaveJsonFile into a file:

program WriteJson;

{$mode objfpc}{$H+}

uses
  SysUtils, fpjson, Pastel.Json;

var
  List: TJSONArray;
  Back: TJSONData;
begin
  List := TJSONArray.Create;
  try
    List.Add(JObject(['name', 'tea', 'price', 2.5, 'hot', True]));
    List.Add(JObject(['name', 'lemonade', 'price', 3, 'hot', False]));
    WriteLn(ToJson(List));
    SaveJsonFile('menu.json', List);           // indented, for people to read too
  finally
    List.Free;
  end;
  Back := LoadJsonFile('menu.json');
  try
    WriteLn(Back.Count, ' items; the second: ', JStr(Back, '[1].name'));
  finally
    Back.Free;
  end;
  DeleteFile('menu.json');
end.

[{"name":"tea","price":2.5,"hot":true},{"name":"lemonade","price":3,"hot":false}]
2 items; the second: lemonade

A TJSONArray owns what is added to it: freeing the array frees the objects inside.

#In Pocket: keeping the entries

Pocket keeps its entries in a JSON file — a list of objects, one per entry:

[
  {"id": 1, "date": "2026-11-03", "kind": "expense", "category": "Food", "amount": 12.5, "note": "Lunch"}
]

Dates are written as yyyy-mm-dd text: readable, sortable, and the same in every country. The kind is written as a word, not as 0 or 1, so the file still makes sense when the enumeration changes. Saving and loading are two procedures:

program PocketFile;

{$mode objfpc}{$H+}

uses
  SysUtils, fpjson, Pastel.Json, PocketMoney, PocketEntries;

procedure SaveEntries(Entries: TEntryList; const FileName: string);
var
  List: TJSONArray;
  E: TEntry;
begin
  List := TJSONArray.Create;
  try
    for E in Entries do
      List.Add(JObject(['id', E.Id, 'date', FormatDateTime('yyyy"-"mm"-"dd', E.Date),
        'kind', KindName(E.Kind), 'category', E.Category, 'amount', Double(E.Amount), 'note', E.Note]));
    SaveJsonFile(FileName, List);
  finally
    List.Free;
  end;
end;

procedure LoadEntries(Entries: TEntryList; const FileName: string);
var
  Data, Item: TJSONData;
  Kind: TEntryKind;
  Y, M, D: Integer;
  S: string;
  E: TEntry;
begin
  Entries.Clear;
  if not FileExists(FileName) then
    Exit;                                  // nothing kept yet
  Data := LoadJsonFile(FileName);
  try
    for Item in JArray(Data, '') do
    begin
      TryParseKind(JStr(Item, 'kind'), Kind);
      S := JStr(Item, 'date');             // '2026-11-03'
      Y := StrToInt(Copy(S, 1, 4));
      M := StrToInt(Copy(S, 6, 2));
      D := StrToInt(Copy(S, 9, 2));
      E := Entries.AddEntry(EncodeDate(Y, M, D), Kind, JStr(Item, 'category'), JFloat(Item, 'amount'),
        JStr(Item, 'note'));
      E.Id := JInt(Item, 'id');
    end;
  finally
    Data.Free;
  end;
end;

var
  Entries: TEntryList;
  E: TEntry;
begin
  Entries := TEntryList.Create;
  try
    Entries.AddEntry(EncodeDate(2026, 11, 3), ekExpense, 'Food', 12.50, 'Lunch').Id := 1;
    Entries.AddEntry(EncodeDate(2026, 11, 1), ekIncome, 'Salary', 3000).Id := 2;
    SaveEntries(Entries, 'pocket.json');
    Entries.Clear;
    WriteLn('Cleared: ', Entries.Count);
    LoadEntries(Entries, 'pocket.json');
    for E in Entries do
      WriteLn(E.Id, ' ', FormatDateTime('yyyy-mm-dd', E.Date), ' ', KindName(E.Kind):8, ' ',
        E.Category:8, ' ', FormatMoney(E.Amount):9, ' ', E.Note);
  finally
    Entries.Free;
  end;
  DeleteFile('pocket.json');
end.

Cleared: 0
1 2026-11-03  expense     Food     12.50 Lunch
2 2026-11-01   income   Salary  3,000.00

New here:

  • JArray(Data, '') — the empty path is the value itself: the list of entries.
  • JFloat reads a number as a Double; assigning it to a Currency rounds it to four decimal places — for amounts with cents, exact again.
  • AddEntry(...).Id := 1 sets a property of the object a function returned, without a variable.
  • KindName(E.Kind):8 — in Write and WriteLn, a : and a number give the width, aligned right. A quick way to make columns.

Chapter 13 turns this pair of procedures into a store: one object that knows where the entries are kept — first this JSON file, later a database — and the rest of Pocket does not need to know which.

#Exercises

  1. Write a program that counts the lines, words and bytes of a text file given as ParamStr(1).
  2. Save a TStringList of names to a file, load it in a second run, and add a name each time.
  3. Load pocket.json and print the total of each kind. Then change a number in the file by hand and run it again.
  4. ★ Make LoadEntries safe against a broken file: catch the exception of LoadJsonFile, and raise one of your own (EStoreError) that names the file and says what was wrong.
