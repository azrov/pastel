#Appendix A. Object Pascal at a glance

The language of part 1 on a few pages, to look things up. The chapter numbers say where each thing is explained.

#A program and a unit (1, 7)

program Name;                      unit Name;
{$mode objfpc}{$H+}                {$mode objfpc}{$H+}
uses SysUtils, Other;              interface
const ...                            uses ...;            what other units see
type ...                             type ... / routine headings
var ...                            implementation
routines                             uses ...;            only for this unit
begin                                the routines' bodies
  statements                       initialization          once, at the start (optional)
end.                               finalization            once, at the end (optional)
                                   end.

#Types (2, 5)

Double, Single real numbers (15 digits, 7)

Boolean True, False

TDateTime a Double: days since 30.12.1899, the time as the fraction
array[1..5] of T, array of T a static array; a dynamic one (SetLength, Length, High; from 0)
record ... end fields together, copied on :=

set of TColor [Red, Blue], in, + - *, Include, Exclude
class ... end an object on the heap, a reference on := (8)

#Operators
+ - * / / always gives a real number
div mod whole-number division and the rest
= <> < > <= >= comparison (= compares; := assigns)
and or not xor logic, and bits for numbers; and/or stop early
shl shr bit shifts
in is in a set
is as is an object of a class; use it as one (8)
@ the address of; a method pointer: @SaveClick

Put comparisons in brackets when they are joined: (A > 0) and (B > 0).

#Routines (4)

#Classes (8)

type
  TShape = class
  private
    FName: string;                     // fields begin with F
    function GetArea: Double; virtual; abstract;
  protected
    procedure Changed; virtual;
  public
    constructor Create(const AName: string);
    destructor Destroy; override;
    property Name: string read FName write FName;
    property Area: Double read GetArea;
  published
    property Caption: string read FName;   // seen by RTTI: .pform files, the designer
  end;

  TCircle = class(TShape)              // inherits from TShape
  ...
    function GetArea: Double; override;
  end;

Obj := TCircle.Create('c');
try
  ...
finally
  Obj.Free;                            // every Create has its Free
end;

inherited calls the parent's method; Self is the object itself; nil is no object; FreeAndNil(Obj) frees and clears the variable.

#Exceptions (9)

try
  ...
except
  on E: EConvertError do ...           // the most specific first
  on E: Exception do WriteLn(E.Message);
end;

raise EMyError.Create('what went wrong');
raise EMyError.CreateFmt('line %d: %s', [N, S]);
raise;                                 // in an except: the same exception again

try ... finally cleans up; try ... except handles. Own exceptions: EMyError = class(Exception);.

#All in one

program Glance;

{$mode objfpc}{$H+}

uses
  SysUtils, Generics.Collections;

type
  TKind = (kExpense, kIncome);

  TItem = class
    Name: string;
    Amount: Currency;
    Kind: TKind;
    constructor Create(const AName: string; AAmount: Currency; AKind: TKind);
    function Signed: Currency;
  end;

  TItems = specialize TObjectList<TItem>;

constructor TItem.Create(const AName: string; AAmount: Currency; AKind: TKind);
begin
  if AAmount < 0 then
    raise EArgumentException.Create('an amount is never negative');
  Name := AName;
  Amount := AAmount;
  Kind := AKind;
end;

function TItem.Signed: Currency;
begin
  if Kind = kIncome then
    Result := Amount
  else
    Result := -Amount;
end;

var
  Items: TItems;
  Item: TItem;
  Total: Currency;
begin
  Items := TItems.Create;
  try
    Items.Add(TItem.Create('Salary', 2400, kIncome));
    Items.Add(TItem.Create('Rent', 650, kExpense));
    Items.Add(TItem.Create('Food', 182.4, kExpense));
    try
      Items.Add(TItem.Create('Oops', -1, kExpense));
    except
      on E: EArgumentException do
        WriteLn('Not added: ', E.Message);
    end;
    Total := 0;
    for Item in Items do
    begin
      WriteLn(Format('%-8s %10.2f', [Item.Name, Double(Item.Signed)]));
      Total := Total + Item.Signed;
    end;
    WriteLn(Format('%-8s %10.2f', ['Balance', Double(Total)]));
  finally
    Items.Free;
  end;
end.

Rent        -650.00
Food        -182.40
